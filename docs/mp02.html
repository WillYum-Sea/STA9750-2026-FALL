The following functions import the data that will be required in this analysis.

This function pulls scrapes the html of the EIA State Electricity Profiles page for each state and extracts the relevant data from the table. It returns a data frame for each state with the following columns: state, abbreviation, CO2_MWh, primary_source, electricity_price_MWh, and generation_MWh. The function also caches the data in the data/mp02 directory to avoid repeated downloads.

get_eia_sep <- function(state, abbr){
    library(tidyverse)
    library(stringr)
    library(httr2)
    library(rvest)
    library(fs)
  
    state_formatted <- str_to_lower(state) |> str_replace_all("\\s", "")
    
    dir_name <- fs::path("data", "mp02")
    file_name <- fs::path(dir_name, state_formatted)
    
    fs::dir_create(dir_name, recurse=TRUE)
    
    Sys.sleep(runif(1, 1, 1.5)) # Random delay between 1 and 1.5 seconds to avoid overwhelming the server
    
    if(!fs::file_exists(file_name)){
        BASE_URL <- "https://www.eia.gov"
        RESPONSE <- request(BASE_URL) |> 
          req_url_path("electricity", 
                       "state", 
                       state_formatted) |>
          req_headers(
            `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0"
            ) |>   
          req_retry(
                max_tries = 4, 
          ) |>
          req_perform(file_name)
    
        resp_check_status(RESPONSE)
    }
    
    TABLE <- read_html(file_name) |> 
        html_element("table") |> 
        html_table() |>
        mutate(Item = str_to_lower(Item))
    
    if("U.S. rank" %in% colnames(TABLE)){
        TABLE <- TABLE |> rename(Rank = `U.S. rank`)
    }
    
    CO2_MWh <- TABLE |> 
        filter(Item == "carbon dioxide (lbs/mwh)") |>
        pull(Value) |> 
        str_replace_all(",", "") |>
        as.numeric()
    
    PRIMARY <- TABLE |> 
        filter(Item == "primary energy source") |> 
        pull(Rank)
    
    RATE <- TABLE |>
        filter(Item == "average retail price (cents/kwh)") |>
        pull(Value) |>
        as.numeric()
    
    GENERATION_MWh <- TABLE |>
        filter(Item == "net generation (megawatthours)") |>
        pull(Value) |>
        str_replace_all(",", "") |>
        as.numeric()
    
    data.frame(CO2_MWh               = CO2_MWh, 
               primary_source        = PRIMARY,
               electricity_price_MWh = RATE * 10, # / 100 cents to dollars &
               # * 1000 kWh to MWH 
               generation_MWh        = GENERATION_MWh, 
               state                 = state, 
               abbreviation          = abbr
    )
}

EIA_SEP_REPORT <- map2(state.name, state.abb, get_eia_sep) |> list_rbind()

load_ntd_energy <- function(){
  library(readxl)
  library(httr2)
  library(tidyverse)
  library(fs)
  
  # Create 'data/mp02' directory if not already present
  DATA_DIR <- fs::path("data", "mp02")
  fs::dir_create(DATA_DIR, recurse=TRUE)

  NTD_ENERGY_FILE <- fs::path(DATA_DIR, "2025_ntd_energy.xlsx")

  if(!fs::file_exists(NTD_ENERGY_FILE)){
    library(httr2)
  
    request("https://www.transit.dot.gov/") |>
      req_url_path("sites",
                   "fta.dot.gov",
                   "files",
                   "2026-09",
                   "2025 Energy Consumption_260806.xlsx") |> 
      req_headers(
        `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0") |> 
      req_perform(NTD_ENERGY_FILE)
  }
  
  if(file.info(NTD_ENERGY_FILE)$size < 1e4){
        cat("I was unable to download the NTD Energy File. Please try again.\n")
        stop("Download failed. Please delete the corrupted file by running `unlink('data/mp02/2025_ntd_energy.xlsx')` and try again.")
    }

  to_numeric_fill_0 <- function(x){
    x <- if_else(x == "-", NA, x)
    replace_na(as.numeric(x), 0)
  }

  read_excel(NTD_ENERGY_FILE) |> 
    select(-c(`Reporter Type`, 
              `Reporting Module`, 
              `Gallons of Other Fuel`, 
              `Other Fuel Description`)) |>
    mutate(across(-c(`Agency Name`, 
                     `Mode`,
                     `TOS`), 
                  to_numeric_fill_0)) |>
    group_by(`NTD ID`, `Mode`, `Agency Name`) |>
    summarize(across(where(is.numeric), sum), 
              .groups = "keep") |>
    mutate(ENERGY = sum(c_across(c(where(is.numeric))))) |>
    filter(ENERGY > 0) |>
    select(-ENERGY) |>
    ungroup()
}

load_ntd_service <- function(){
  library(httr2)
  library(tidyverse)
  library(fs)
  
  # Create 'data/mp02' directory if not already present
  DATA_DIR <- fs::path("data", "mp02")
  fs::dir_create(DATA_DIR, recurse=TRUE)

  NTD_SERVICE_FILE <- fs::path(DATA_DIR, "2025_ntd_service.csv")

  if(!fs::file_exists(NTD_SERVICE_FILE)){
    library(httr2)
  
    request("https://data.transportation.gov/resource/4fir-qbim.csv") |>
      req_url_query(`$query`=paste(
        "SELECT  `_5_digit_ntd_id` AS `nta_id`", 
        "`max_agency` AS `agency`",
        "`max_reporter_type` AS `reporter_type`",
        "`max_organization_type` AS `organization_type`",
        "`max_city` AS `city`",
        "`max_state` AS `state`",
        "`mode`",
        "`max_mode_name`",      #for the join to recode mode names in data cleaning section
        "`type_of_service`",
        "`max_primary_uza_code` as `uza_code`",
        "`max_primary_uza_name` as `uza_name`",
        "`max_primary_uza_area_sq_miles` as `uza_sq_miles`",
        "`max_primary_uza_population` as `uza_population`",
        "`max_service_area_sq_miles` as `service_area_sq_miles`",
        "`max_service_area_population` as `service_area_population`",
        "`questionable_record` as `flagged_likely_inaccurate`",
        "`sum_unlinked_passenger_trips_upt` as `total_upt`",
        "`sum_passenger_miles` WHERE caseless_one_of(`report_year`, '2025') limit 50000", 
        sep=",")) |>
      req_headers(
        `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0") |> 
      req_perform(NTD_SERVICE_FILE)
  }
  
  if(file.info(NTD_SERVICE_FILE)$size < 1e4){
        cat("I was unable to download the NTD Service File. Please try again.\n")
        stop("Download failed. Please delete the corrupted file by running `unlink('data/mp02/2025_ntd_service.xlsx')` and try again.")
    }

  read_csv(NTD_SERVICE_FILE) |>
    mutate(flagged_likely_inaccurate = !is.na(flagged_likely_inaccurate))
}

The following pulls the data from the EIA CO2 Emissions Factors table from the EIA website. The function scrapes the HTML of the page and extracts the relevant data from the table. The function also caches the data in the data/mp02 directory to avoid repeated downloads. This will be used later in the analysis.

Code 
load_eia_co2_factors <- function() {

  library(httr2)
  library(rvest)
  library(tidyverse)
  library(fs)
  library(stringr)

  # Create data storage directory
  DATA_DIR <- fs::path("data", "mp02")
  fs::dir_create(DATA_DIR, recurse = TRUE)

  # Local cache file
  EIA_CO2_FILE <- fs::path(
    DATA_DIR,
    "2024_eia_co2_factors.csv"
  )

  # Download and process only if file does not exist
  if (!fs::file_exists(EIA_CO2_FILE)) {

    PAGE <- request(
      "https://www.eia.gov/environment/emissions/co2_vol_mass.php"
    ) |>
      req_headers(
        `User-Agent` =
          "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0"
      ) |>
      req_perform() |>
      resp_body_html()

    # Extract first HTML table
    CO2_TABLE <- PAGE |>
      html_element("table") |>
      html_table()

    # Assign column names manually
    names(CO2_TABLE) <- c(
      "fuel",
      "lb_co2_per_unit",
      "kg_co2_per_unit",
      "lb_co2_per_mmbtu",
      "kg_co2_per_mmbtu"
    )

    # Remove category headings and footnotes
    CO2_TABLE <- CO2_TABLE |>
      filter(
        !fuel %in% c(
          "For homes and businesses",
          "Other transportation fuels",
          "Industrial fuels and others not listed above",
          "Nonfuel uses",
          "Coals by type",
          "Other fuels"
        )
      ) |>
      filter(
        !str_detect(
          fuel,
          "^Data source:"
        )
      )

    # Extract numeric values from columns that contain units
    CO2_TABLE <- CO2_TABLE |>
      mutate(
        across(
          -fuel,
          ~ str_extract(
            as.character(.x),
            "[0-9,.]+"
          )
        )
      ) |>
      mutate(
        across(
          -fuel,
          ~ as.numeric(
            str_remove_all(.x, ",")
          )
        )
      )

    # Save cleaned table locally
    write_csv(
      CO2_TABLE,
      EIA_CO2_FILE
    )
  }

  # Verify successful download
  if (file.info(EIA_CO2_FILE)$size < 100) {

    cat(
      "I was unable to download the EIA CO2 factors table.\n"
    )

    stop(
      "Download failed. Delete the corrupted file with ",
      "`unlink('data/mp02/eia_co2_factors.csv')` ",
      "and try again."
    )
  }

  # Read cleaned table
  read_csv(
    EIA_CO2_FILE,
    show_col_types = FALSE
  )
}

EIA_CO2_FACTORS <- load_eia_co2_factors()

#Alternative method of generating a 'dirtiness' score for each state based on 
#electricity carbon intensity and price. This method uses min-max normalization 
#to scale both CO2/MWh and electricity price ($/MWh) between 0 and 1, and then 
#averages the two normalized values to create a composite 'dirty score'. Higher 
#scores indicate states with electricity that is both relatively more 
#carbon-intensive and more expensive. This score can be interpreted as 
#"How unfavorable are the environmental and economic conditions for 
#electrication compared to other states?" It does not highlight where 
#decarbonization efforts are most needed, but rather where electricity is both 
#relatively more carbon-intensive and more expensive.

EIA_SEP_REPORT_SCORE <-
  EIA_SEP_REPORT |>
  mutate(
    co2_norm =
      (CO2_MWh - min(CO2_MWh, na.rm = TRUE)) /
      (max(CO2_MWh, na.rm = TRUE) -
         min(CO2_MWh, na.rm = TRUE)),

    price_norm =
      (electricity_price_MWh -
         min(electricity_price_MWh, na.rm = TRUE)) /
      (max(electricity_price_MWh, na.rm = TRUE) -
         min(electricity_price_MWh, na.rm = TRUE)),

    dirty_score =
      (co2_norm + price_norm) / 2
  )

The first table displayed simply ranks the states by carbon intensity of electricity generation (CO2/MWh). The second table introduces a composite ‘dirty score’ that equally weights both carbon intensity and electricity price, providing a more holistic view of the environmental and economic conditions for electrification in each state. For example, a state with a clean electric grid will reduce emissions greatly via electrification because it will eliminate diesel emissions by contributing comparatively less emissions via their already clean grid. A state with a dirtier grid will still benefit from electrification by reducing tailpipe diesel emissions, but to a lesser extent because more carbon will be emitted by their comparatively carbon-intensive grid to provide the necessary electricity to electrify transit.

There is some overlap between the tables with West Virginia and Hawaii. This is likely due to West Virginia’s historic relationship with coal power and Hawaii’s reliance on imported oil for electricity generation. Both states have relatively high electricity prices and carbon intensity, which is reflected in their rankings in both tables.

On average, how many pounds of CO2 are emitted per MWh of electricity produced in the US?

Code 
AVG_CO2_MWh <- EIA_SEP_REPORT |>
  summarize(
    total_generation = sum(generation_MWh, na.rm = TRUE),
    total_weighted_CO2 = sum(generation_MWh * CO2_MWh, na.rm = TRUE),
    avg_CO2_MWh = total_weighted_CO2 / total_generation
  )
pull(AVG_CO2_MWh)

Using this equation: \[ \text{Weighted Average CO}_2\text{ per MWh} = \frac{ \sum_{i=1}^{n} (\text{Generation}_i \times \text{CO}_2\text{/MWh}_i) }{ \sum_{i=1}^{n} \text{Generation}_i } \] We can calculate that the average C02 emissions per MWh of electricity generation in the United States is approximately 784.9 lbs.

#creating table for both analyses

#This table produces 2 instances of many-to-many relationships, but it does not 
#matter for the purposes of this analysis because we are only interested in the 
#total emissions and total UPT for each agency-mode combination. The 
#many-to-many relationships will not affect the totals, but it is important to 
#note that this is not a clean join and should be treated with caution if used 
#for other purposes. The 'size_category' column is created by dividing the 
#agencies into three equal groups based on their total passenger miles. This is 
#done using the 'ntile' function, which assigns a rank to each agency based on 
#their total passenger miles and then divides them into three groups 
#(small, medium, large). The resulting table includes the total CO2 emissions, 
#total UPT, total passenger miles, CO2 emissions per capita, and size category 
#for each agency-mode combination. For the purposes of this analysis, transit
#agencies with fewer than 1000 UPT are filtered out

EMISSIONS_PER_CAPITA <- FINAL_DATA |>
  left_join(
    NTD_SERVICE |>
      select(agency, mode, total_upt, sum_passenger_miles),
    by = c("agency", "mode")
  ) |>
  mutate(
    emissions_per_upt = total_co2_output / total_upt,
    emissions_per_passenger_mile = total_co2_output / sum_passenger_miles
  ) |>
  mutate(
    size_category = c("Small", "Medium", "Large")[
      ntile(sum_passenger_miles, 3)
    ]
  ) |> 
  select(
    c(
      agency,      
      mode,
      total_co2_output,
      total_upt,
      sum_passenger_miles,
      emissions_per_upt,
      emissions_per_passenger_mile,
      size_category)
    )|>
    drop_na(c(total_co2_output, total_upt, sum_passenger_miles)) |>
  filter(total_upt > 1000 )

#summarizing EMISSIONS_PER_CAPITA to represent all modes by agency
EMISSIONS_PER_CAPITA_AGENCY <- EMISSIONS_PER_CAPITA |>
  group_by(agency) |>
  summarize(
    total_co2_output = sum(total_co2_output, na.rm = TRUE),
    total_upt = sum(total_upt, na.rm = TRUE),

    .groups = "drop"

    emissions_per_upt = total_co2_output / total_upt,

      ntile(passenger_miles, 3) == 2 ~ "Medium",
      ntile(passenger_miles, 3) == 3 ~ "Large"
    )
  )

Transit Agency Emissions Researcher Awards “Greenest Transit” Title to McKinney Avenue Transit Authority: A Small Streetcar with a Big Impact

Dallas, TX — A new analysis of public transit agencies across the United States found substantial variation in carbon emissions per passenger trip.   • While large transit agencies benefit from economies of scale and higher rider volumes smaller agencies can achieve low emissions per trip through efficient operations and clean energy sources.
  • Significant differences existed between agencies of similar size due to variations in fuel mix, ridership patterns, and regional electricity generation portfolios.
  • Accounting for electricity-related emissions materially changed the emissions rankings of many transit agencies.

Methodology

The analysis combined multiple federal datasets, including:

  • National Transit Database fuel and energy consumption records.
  • National Transit Database service and ridership statistics.
  • U.S. Energy Information Administration electricity generation data.
  • U.S. Energy Information Administration carbon dioxide emissions factors.

Agency-level emissions were calculated by converting fuel and electricity consumption into carbon emission equivalents and aggregating results across all modes of transit operated by each agency. Agencies were subsequently categorized into small, medium, and large groups based on total passenger miles traveled.

Implications

The results suggest that transit decarbonization strategies should consider both vehicle technology and regional electricity generation characteristics. Investments in grid decarbonization may be as important as fleet electrification for achieving substantial emissions reductions.
